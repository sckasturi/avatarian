/* Avatarian renderer. Credits and sources: see AVATARIAN.md in the project repo. */
/* Avatarian gadget styling — paste into MediaWiki:Common.css
   Blocks are PAIRS of phonemes, written top slot then bottom slot, and
   packed tight (see the header of js/render.js). Sizing follows the
   SOUND, not the slot — consonants 5 units tall, vowels 4, stacked
   flush — so a vowel is 4/5 of 1.25em in HEIGHT but keeps the full
   1.25em width: canon draws vowel marks wide and flat, spanning their
   partner. The vowel gets that shape from a separately generated flat
   drawing (100x80, see tools/build_glyphs.py), so it is scaled
   uniformly and its stroke weight matches the consonants.
   Keep these values in step with site/css/style.css; both drive the
   same render.js.
   Sized in em so Avatarian scales with the surrounding wiki text,
   and coloured via currentColor so it matches link/body text. */
/* A nine-row block is 1.64 x a consonant (1.25em) = 2.05em tall, so at
   the surrounding text size it stands about twice as tall as the words
   around it. Scaling the whole word's font-size by 1/2.05 makes the whole
   glyph STACK equal the font size — the block sits on the text line
   instead of towering over it. Everything inside is sized in em, so this
   one knob scales it all. Want the glyphs a little bigger or smaller?
   lower or raise the 2.05 (e.g. / 1.6 for taller, / 2.5 for shorter). */
.avatarian-word {
  display: inline-flex; align-items: flex-start; vertical-align: middle;
  margin: 0 0.3em 0 0;
  font-size: calc(1em / 2.05);
}
/* No flash of English before the glyphs: hide the plain fallback text
   until the gadget has drawn a word (it sets data-avatarian-done on each).
   Scoped to .client-js — MediaWiki adds that to <html> before first paint,
   so a reader with JavaScript OFF still sees the text, and the loader
   reveals anything still undrawn after a few seconds as a safety net. */
html.client-js .avatarian-word:not([data-avatarian-done]) { visibility: hidden; }
/* The glyphs are SVG, which the browser's text selection never highlights,
   so a word caught in a selection would sit un-tinted between highlighted
   words. The gadget adds this class to any word the selection touches; a
   translucent blue reads as "selected" on a light or dark background. */
.avatarian-word.avatarian-selected { background: rgba(51, 144, 255, 0.32); }
/* Each `/`-separated word in one {{Avatarian}} is drawn as its own part,
   so words keep a gap between them (the outer span rows the parts;
   inline-flex here rows each part's blocks). The gap reads as a word
   space at the scaled size. */
.avatarian-word-part { display: inline-flex; align-items: flex-start; }
.avatarian-word-part + .avatarian-word-part { margin-left: 0.5em; }
.avatarian-block { display: inline-flex; flex-direction: column; align-items: center; margin-right: 0.05em; }
.avatarian-slot { display: flex; justify-content: center; }
/* The two glyphs in a block share a lattice edge (glyphspec.py: "stack
   flush, 5 units over 4"), but each SVG carries its own clearance margin
   outside the lattice — 10svg on a 100 box for a consonant (0.125em),
   8svg on an 80 box for a flat vowel (0.1em). Pull the bottom slot up by
   their sum so the lattices meet instead of leaving both margins as a
   gap. Keep in step with site/css/blocks.css, where the same pull-up is
   calc(var(--av-size) * -0.18) — -0.18 of a 1.25em consonant is 0.225em. */
.avatarian-slot-bottom { margin-top: -0.225em; }
.avatarian-glyph { display: inline-block; line-height: 0; color: currentColor; }
.avatarian-glyph svg { display: block; width: 100%; height: 100%; }
/* Scaled down to the font size, the default stroke (9 of 100 lattice units)
   renders around a pixel wide and reads faint. Thicken it — the lattice is
   the same scale for letters and marks (80 units per em), so one value
   fits both. 13 is a readable balance; raise for bolder, but too high and
   adjacent strokes start to merge. Wiki only; the standalone site draws
   glyphs large enough that 9 is fine. */
.avatarian-glyph svg,
.avatarian-mark svg { stroke-width: 13; }
.avatarian-glyph.avatarian-consonant { width: 1.25em; height: 1.25em; }
.avatarian-glyph.avatarian-null_consonant { width: 1.25em; height: 1.25em; }
.avatarian-glyph.avatarian-vowel { width: 1.25em; height: 1em; }
.avatarian-glyph.avatarian-null { width: 1.25em; height: 1em; }
/* A block of two consonants (B1, settled session 12): two full-size
   consonants OVERLAPPING by one lattice row — the top glyph's bottom row
   is the bottom glyph's top row — so ten rows of content total nine.
   Both stay full size, so stroke weight matches the rest. Two 1.25em
   boxes stacked flush leave a 0.25em gap between their lattices (0.125em
   clearance each); closing it and overlapping one more row (0.2em) is
   0.45em. Replaces the -0.225em the V-C bottom slot uses.

   render.js tags a two-consonant block `avatarian-cc`, so this is a plain
   class rather than a :has() selector — :has() isn't in older browsers
   and MediaWiki's CSS linter rejects it. */
.avatarian-block.avatarian-cc .avatarian-slot-bottom { margin-top: -0.45em; }
/* A 3-row vowel is drawn bottom-aligned in its 4-row box, which is right
   in the bottom slot and wrong in the top: there the empty row lands at
   the block's outer edge as dead space and the vowel sits flush against
   the consonant. Pull it up one lattice row so the gap
   falls between the two glyphs — rows 1-3 vowel, row 4 gap, rows 5-9
   consonant. One lattice row is 20% of the box, not 25%: the viewBox is
   80 units, 64 of lattice plus 16 of clearance margin outside it.
   4-row vowels fill their box and abut directly, so they are
   excluded. */
.avatarian-slot-top .avatarian-glyph.avatarian-vowel:not(.avatarian-4row),
.avatarian-slot-top .avatarian-glyph.avatarian-null:not(.avatarian-4row) {
  transform: translateY(-20%);
}
/* Bottom-slot glyphs listed in FLIPS are the same drawing mirrored
   top-to-bottom. On the SVG, so it composes with the row shift above
   rather than overwriting it. */
.avatarian-glyph.avatarian-flipped svg { transform: scaleY(-1); }
/* A 3-row vowel that also flips (ae, schwa, ai) is drawn bottom-aligned
   with its empty row on top; mirroring it for the bottom slot pushes that
   row to the block's outer edge, so the ink floats up off the baseline.
   Shift it back down one lattice row — the counterpart of the top-slot
   pull-up above. On the span, so it composes with the SVG's scaleY(-1). */
.avatarian-slot-bottom .avatarian-glyph.avatarian-vowel.avatarian-flipped:not(.avatarian-4row),
.avatarian-slot-bottom .avatarian-glyph.avatarian-null.avatarian-flipped:not(.avatarian-4row) {
  transform: translateY(20%);
}
/* Vowels ship a square (100x100) and a flat (100x80) SVG; use the flat
   one. Consonants have only the square, so :not(:only-of-type) leaves it. */
.avatarian-glyph .g-square:not(:only-of-type) { display: none; }
.avatarian-placeholder { opacity: 0.5; }
.avatarian-missing { font-size: 0.6em; color: #b32; line-height: 1; }
/* Punctuation marks: a THIRD height class — one lattice column wide (a
   question mark is two) and nine rows tall, standing beside the writing
   rather than in a slot. The box is 36x164 units (52x164 for the wide
   question mark) against a letter's 100x100, so a mark is 164/100 = 1.64
   of a consonant's height. render.js sets each mark's aspect-ratio inline
   from its own viewBox, so only the height is fixed here and a wider mark
   keeps its true proportions. Keep in step with site/css/blocks.css. */
.avatarian-mark {
  display: inline-block; line-height: 0; color: currentColor;
  align-self: stretch;
  height: calc(1.25em * 1.64);
}
.avatarian-mark svg { display: block; width: 100%; height: 100%; }
/* The "*" unreadable-glyph marker: dimmed, and never flipped. Two classes
   + svg matches the flip rule's specificity and comes later, so it wins
   without !important. */
.avatarian-unreadable { opacity: 0.68; }
.avatarian-glyph.avatarian-unreadable svg { transform: none; }
